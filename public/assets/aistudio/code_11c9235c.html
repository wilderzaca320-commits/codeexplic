@idempotent_handler(ttl_seconds=86400, key_header="Stripe-Signature")
async def handle_stripe_event(event_payload: dict, db: AsyncSession):
    event_id = event_payload.get("id")
    if await redis_pool.exists(f"lock:payment:{event_id}"):
        return JSONResponse(content={"status": "processing_locked"}, status_code=200)

    # Transacción atómica protegida con Saga rollback
    async with db.begin():
        saga = PaymentSaga(event_id)
        await saga.execute_transition(status=PaymentStatus.CONFIRMED)
        await kafka_producer.publish("orders.paid", event_payload)
