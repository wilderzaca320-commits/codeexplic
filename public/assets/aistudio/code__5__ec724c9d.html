04 async processChargeSucceeded(event: StripeEvent): Promise<void> {
05 const lock = await this.redlock.acquire([`lock:${event.id}`], 5000);
06 try {
07 const order = await this.ordersRepo.findOneOrFail({ where: { txId: event.id } });
08 this.eventEmitter.emit('order.settled', { orderId: order.id });
09 } finally {
10 await lock.release();
11 }
12 }
13 }
1

Garantía de Idempotencia con Redlock (Línea 05)

La función previene condiciones de carrera causadas por reintentos automáticos del webhook de Stripe implementando un bloqueo atómico distribuido en Redis con un TTL de seguridad de 5,000 milisegundos.

2

Búsqueda Estricta de Orden (Línea 07)

Utiliza el operador estricto findOneOrFail para arrojar de manera inmediata una excepción si no existe el identificador de transacción en PostgreSQL, abortando el ciclo sin disparar eventos de dominio.

3

Desacoplamiento Mediante Eventos (Línea 08)

La orden no modifica inventario directamente aquí; en su lugar, delega la actualización publicando el evento asíncrono 'order.settled' hacia el bus de eventos del microservicio de facturación.
